def ship_cost(order):
 if order is None:raise ValueError('no order')
 if order['qty']<=0:raise ValueError('qty must be positive')
 us=order['country']=='US'
 heavy=order['weight']>10
 if us:cost=20 if heavy else 10
 if not us:cost=45 if heavy else 30
 if order.get('express'):cost=cost*2
 return cost
