  R new RangeError('n must be positive')

 ?m.length===0
  r []
 k out:number[][]=[]

  k row:number[]=[]
  L l i=0;i<m.length;i++

 k out:number[][]=[]

  k row:number[]=[]
  L l j=0;j<bCols;j++
